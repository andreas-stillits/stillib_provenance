""" 
Example config.py file based on Pydantic 2.


- Any Config class inherits from Pydantic's regular BaseModel class.

- Parameters can be declared using regular light weight type hinting or
  for a more comprehensive definition and additional constraints on valid inputs
  using the Field class - see examples

A project can have several config classes for different programs.
These can be declared separately or alternatively nested into each other.

"""

from pydantic import BaseModel, ConfigDict, Field


# Example config class definition
class Config(BaseModel):
    
    # define config behavior - always include this line
    model_config = ConfigDict(extra="forbid", frozen=True, validate_default=True)

    # Examples of minimal declaration:

    diffusivity: float = 1.0
    n_steps: int = 1000
    ignore_decay: bool = False
    principal_direction: tuple[float, float, float] = (0.0, 0.0, 1.0)


"""
    # Examples of Field declaration with constraints and additional information
    diffusivity: float = Field(
        1.0,
        gt = 0, # enforces that the diffusivity must be grater than zero / a positive number to be acceptable
        description = "Diffusion coefficient [µm²/s]"
    )

    n_steps: int = Field(
        1000,
        ge=1, # enforces that the n_steps must be greater or equal to 1 to be acceptable
        description = "Number of simulation steps"
    )

    ignore_decay: bool = Field(

    )

    principal_direction: tuple[float, float, float] = Field(
